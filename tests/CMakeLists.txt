find_package(GTest REQUIRED)
include(GoogleTest)
file(GLOB ALG_EXP_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_executable(algorithm-explorer-tests ${ALG_EXP_TEST_SOURCES})
target_link_libraries(algorithm-explorer-tests PRIVATE algorithm-explorer algorithm-explorer-warnings GTest::gtest GTest::gtest_main)
find_program(Z3_EXECUTABLE z3)
if(Z3_EXECUTABLE)
  target_compile_definitions(algorithm-explorer-tests PRIVATE ALG_EXP_Z3_PATH="${Z3_EXECUTABLE}")
endif()
gtest_discover_tests(algorithm-explorer-tests DISCOVERY_TIMEOUT 60)
