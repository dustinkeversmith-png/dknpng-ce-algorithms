cmake_minimum_required(VERSION 3.24)
project(combinatorial VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(COMBINATORIAL_BUILD_TESTS "Build unit tests" ON)
option(COMBINATORIAL_BUILD_BENCHMARKS "Build Google Benchmark targets" ON)
option(COMBINATORIAL_BUILD_TOOLS "Build the CLI driver" ON)
option(COMBINATORIAL_BUILD_FUZZERS "Build fuzz harnesses" ON)
option(COMBINATORIAL_SANITIZE "Build with ASan + UBSan" OFF)

add_library(combinatorial_warnings INTERFACE)
target_compile_options(combinatorial_warnings INTERFACE -Wall -Wextra -Wpedantic -Werror)

if(COMBINATORIAL_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
endif()

file(GLOB_RECURSE COMBINATORIAL_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(combinatorial STATIC ${COMBINATORIAL_SOURCES})
add_library(combinatorial::combinatorial ALIAS combinatorial)
target_include_directories(combinatorial PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(combinatorial PRIVATE combinatorial_warnings)

if(COMBINATORIAL_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(COMBINATORIAL_BUILD_BENCHMARKS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/bench/CMakeLists.txt)
  add_subdirectory(bench)
endif()
if(COMBINATORIAL_BUILD_TOOLS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/CMakeLists.txt)
  add_subdirectory(tools)
endif()
if(COMBINATORIAL_BUILD_FUZZERS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/fuzz/CMakeLists.txt)
  add_subdirectory(fuzz)
endif()
