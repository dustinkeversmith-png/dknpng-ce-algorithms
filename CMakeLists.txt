cmake_minimum_required(VERSION 3.24)
project(algorithm-explorer VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(ALG_EXP_BUILD_TESTS "Build unit tests" ON)
option(ALG_EXP_BUILD_BENCHMARKS "Build Google Benchmark targets" ON)
option(ALG_EXP_BUILD_TOOLS "Build the CLI driver" ON)
option(ALG_EXP_BUILD_FUZZERS "Build fuzz harnesses" ON)
option(ALG_EXP_SANITIZE "Build with ASan + UBSan" OFF)

add_library(algorithm-explorer-warnings INTERFACE)
target_compile_options(algorithm-explorer-warnings INTERFACE -Wall -Wextra -Wpedantic -Werror)

if(ALG_EXP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  add_link_options(-fsanitize=address,undefined)
endif()

file(GLOB_RECURSE ALG_EXP_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(algorithm-explorer STATIC ${ALG_EXP_SOURCES})
add_library(algorithm-explorer::algorithm-explorer ALIAS algorithm-explorer)
target_include_directories(algorithm-explorer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(algorithm-explorer PRIVATE algorithm-explorer-warnings)

if(ALG_EXP_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(ALG_EXP_BUILD_BENCHMARKS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/bench/CMakeLists.txt)
  add_subdirectory(bench)
endif()
if(ALG_EXP_BUILD_TOOLS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/CMakeLists.txt)
  add_subdirectory(tools)
endif()
if(ALG_EXP_BUILD_FUZZERS AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/fuzz/CMakeLists.txt)
  add_subdirectory(fuzz)
endif()
