find_package(GTest REQUIRED)
include(GoogleTest)
file(GLOB COMBINATORIAL_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.cpp)
add_executable(combinatorial_tests ${COMBINATORIAL_TEST_SOURCES})
target_link_libraries(combinatorial_tests PRIVATE combinatorial combinatorial_warnings GTest::gtest GTest::gtest_main)
find_program(Z3_EXECUTABLE z3)
if(Z3_EXECUTABLE)
  target_compile_definitions(combinatorial_tests PRIVATE COMBINATORIAL_Z3_PATH="${Z3_EXECUTABLE}")
endif()
gtest_discover_tests(combinatorial_tests DISCOVERY_TIMEOUT 60)
